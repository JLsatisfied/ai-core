// RSA 公钥加密流程
1. publicKey()        → GET /aicg/user/publicKey   获取 RSA 公钥

Axios 封装 ​

ts
// utils/http/axios.ts
const httpDefault = axios.create({
  baseURL: import.meta.env.VITE_APP_API_URL || '/aicg/',
  timeout: 30000,
})

// 请求拦截器 — 自动附加 token
httpDefault.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

// 响应拦截器 — 401 自动跳转登录
httpDefault.interceptors.response.use(
  (res) => res,
  (err) => {
    if (err.response?.status === 401) {
      localStorage.removeItem('token')
      router.push('/')
    }
    return Promise.reject(err)
  },
)

// FormData 实例 (用于文件上传)
const httpFormData = axios.create({
  baseURL: '/aicg/',
  headers: { 'Content-Type': 'multipart/form-data' },
})

Pinia 状态管理 ​

ts
// store/index.ts
export const useStore = defineStore('user-storeId', {
  state: () => ({
    token: localStorage.getItem('token') as string | null,
    freeEntry: localStorage.getItem('freeEntry') as string | null, // 30天免登录
  }),
  actions: {
    setToken(newToken: string, persist = false) {
      this.token = newToken
      if (persist) localStorage.setItem('freeEntry', 'true')
      localStorage.setItem('token', newToken)
    },
    clearToken() {
      this.token = ''
      this.freeEntry = ''
      localStorage.removeItem('token')
      localStorage.removeItem('freeEntry')
    },
  },
})

组件通信模式 ​

整个 AI 对话页面采用单一大组件 + Props/Emits 子组件模式：

intelligence/index.vue (1300+ 行，编排者)
  │
  ├── Props ↓         Emits ↑
  ├─▶ SideBar          ← selectMessage, addClassify, removeClassify
  ├─▶ NavBar           ← switchModel, openUserProfile, outLogin
  ├─▶ ChatInput        ← send, stopExport, toggleDeepThinking, parseFileInput, ...
  ├─▶ ChatMessageItem  ← editChat, copyDocument, speak, toggleThink
  ├─▶ TemplateDialog   ← v-model:visible, selectPrompt, citeTemplate, pageChange
  └─▶ UserProfileModal ← close, save, selectFile, uploadImg

所有状态和业务逻辑集中在 index.vue 中管理，子组件仅负责 UI 展示和事件触发。

主线程                               Worker 线程
  │                                      │
  │  懒加载 new Worker()                  │
  │  ─────────────────────────────────▶  │

  │                                      │
  │  ◀─────────────────────────────────  │
  │  postMessage(rendered_html)           │
  │                                      │
  ▼  v-html 渲染                          │
  ▼  事件代理处理复制/下载                  │
