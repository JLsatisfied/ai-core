路由守卫 ​

ts
// router/index.ts
import NProgress from 'nprogress'
import 'nprogress/nprogress.css'

const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: '/', name: 'HomePage', component: () => import('@/view/home/index.vue') },
    { path: '/intelligence', name: 'IntelligencePage', component: () => import('@/view/intelligence/index.vue') },
    { path: '/adventure', name: 'AdventurePage', component: () => import('@/view/adventure/index.vue') },
    { path: '/customer-service', name: 'CustomerService', component: () => import('@/view/customer-service/index.vue') },
    { path: '/:pathMatch', name: '404Page', component: () => import('@/view/error/404.vue') },
  ],
})

// 全局前置守卫 — 鉴权 + 进度条
router.beforeEach((to) => {
  NProgress.start()
  const store = useStore()
  if (!store.token && to.path !== '/') {
    ElMessage.warning('请先登录')
    return { name: 'home' }
  }
})

// 全局后置守卫 — 完成进度条
router.afterEach(() => {
  NProgress.done()
})

所有页面路由均使用懒加载 (() => import(...))，减小首屏包体积。

页面结构 ​

首页 (/) ​

像素 RPG 风格登录/注册页面，包含四种模式：

  • LOGIN — 账号密码登录，支持"30天免登录"
  • REGISTER — 新用户注册（昵称/账号/密码/确认密码/邮箱/验证码/头像）
  • NEW CODE — 修改密码（账号 + 原密码 + 新密码）
  • RESET PASSWORD — 重置密码（通过邮箱）

所有密码在传输前使用 RSA 公钥加密。验证码由后端生成 SVG，前端内联渲染，点击可刷新。

NProgress 进度条 ​

所有路由跳转都会触发顶部的 NProgress 进度条动画：

  • router.beforeEach → NProgress.start() — 开始加载动画
  • router.afterEach → NProgress.done() — 加载完成
